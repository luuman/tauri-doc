cst-web-meeting​

// 开发模式 localhost:
npm run start

-- /src
	-- /assets          img,icon,css 等
	-- /service         业务逻辑代码
	-- /slice           全局状态存储与修改
	-- /library         依赖的其他资源
	-- /routes          路由
	-- /components      组件
		-- /base            基础组件
		-- /meeting         会议组件
		-- /room            房间组件
	-- /pages           页面
	-- /hooks           react hook
	-- /encryption      加解密
	-- /http            http请求
	-- /utils           通用工具类
	-- /constant        常量
	-- /config          配置项
	-- /window-mount    挂载到window上的方法
// 打包
npm run build

// proto 编译js
npm run proto

// proto 编译ts
npm run protots
